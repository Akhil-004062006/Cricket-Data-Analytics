# Load CSV and JSON files
def load_all_files(folder):

    datasets = {}

    files = glob.glob(
        os.path.join(folder, "**", "*"),
        recursive=True
    )

    for file_path in files:

        if not os.path.isfile(file_path):
            continue

        if OUTPUT_FOLDER in os.path.abspath(file_path):
            continue

        file_name = os.path.basename(file_path)
        extension = Path(file_path).suffix.lower()
        dataset_name = Path(file_name).stem.lower()

        try:

            if extension == ".csv":

                df = pd.read_csv(file_path)

            elif extension == ".json":

                with open(file_path, "r", encoding="utf-8") as file:
                    json_data = json.load(file)

                if isinstance(json_data, list):
                    df = pd.json_normalize(json_data)

                elif isinstance(json_data, dict):

                    if isinstance(json_data.get("data"), list):
                        df = pd.json_normalize(json_data["data"])

                    elif isinstance(json_data.get("records"), list):
                        df = pd.json_normalize(json_data["records"])

                    else:
                        df = pd.json_normalize(json_data)

                else:
                    continue

            else:
                continue

            datasets[dataset_name] = df

            print(
                "Loaded:", file_name,
                "| Rows:", len(df)
            )

        except Exception as e:
            print("Error loading", file_name, ":", e)

    return datasets


datasets = load_all_files(DATA_FOLDER)


# Identify datasets using column names
def find_dataset(required_columns):

    for name, df in datasets.items():

        if set(required_columns).issubset(set(df.columns)):
            print("Using dataset:", name)
            return df.copy()

    return None

# Data cleaning
def clean_dataframe(df):

    if df is None:
        return None

    df = df.copy()
    df.columns = df.columns.str.strip()
    df = df.drop_duplicates()

    for column in df.select_dtypes(include="object").columns:

        df[column] = df[column].apply(
            lambda x: x.strip() if isinstance(x, str) else x
        )

    return df


match_df = clean_dataframe(match_df)
batting_df = clean_dataframe(batting_df)
bowling_df = clean_dataframe(bowling_df)


# Convert numeric columns
def convert_numeric(df, columns):

    if df is None:
        return

    for column in columns:

        if column in df.columns:

            df[column] = pd.to_numeric(
                df[column],
                errors="coerce"
            ).fillna(0)


convert_numeric(
    batting_df,
    ["runs", "balls", "4s", "6s", "SR", "battingPos"]
)

convert_numeric(
    bowling_df,
    ["wickets", "runs", "overs", "maidens", "balls"]
)

# Batting Analysis
if batting_df is not None:

    batting_stats = batting_df.groupby(
        "batsmanName"
    ).agg(
        total_runs=("runs", "sum"),
        total_balls=("balls", "sum"),
        total_fours=("4s", "sum"),
        total_sixes=("6s", "sum"),
        innings=("match_id", "count")
    ).reset_index()

    batting_stats["strike_rate"] = np.where(
        batting_stats["total_balls"] > 0,
        batting_stats["total_runs"] /
        batting_stats["total_balls"] * 100,
        0
    )

    batting_stats["average_runs"] = np.where(
        batting_stats["innings"] > 0,
        batting_stats["total_runs"] /
        batting_stats["innings"],
        0
    )

    batting_stats["consistency_index"] = (
        batting_df.groupby("batsmanName")["runs"]
        .std()
        .reindex(batting_stats["batsmanName"])
        .fillna(0)
        .values
    )

    batting_stats["power_factor"] = (
        batting_stats["total_sixes"] * 6 +
        batting_stats["total_fours"] * 4
    )

    print("\nBatting Statistics")
    display(batting_stats.head(10))

# Bowling Analysis
if bowling_df is not None:

    bowling_stats = bowling_df.groupby(
        "bowlerName"
    ).agg(
        total_wickets=("wickets", "sum"),
        innings=("match_id", "count")
    ).reset_index()

    if "runs" in bowling_df.columns:

        bowling_runs = bowling_df.groupby(
            "bowlerName"
        )["runs"].sum()

        bowling_stats["runs_conceded"] = (
            bowling_stats["bowlerName"].map(bowling_runs)
        )

    if "overs" in bowling_df.columns:

        def overs_to_balls(value):

            try:

                parts = str(value).split(".")

                completed_overs = int(parts[0])

                extra_balls = (
                    int(parts[1]) if len(parts) > 1 else 0
                )

                if extra_balls > 5:
                    return 0

                return completed_overs * 6 + extra_balls

            except:
                return 0

        bowling_df["balls_bowled"] = (
            bowling_df["overs"].apply(overs_to_balls)
        )

        total_balls = bowling_df.groupby(
            "bowlerName"
        )["balls_bowled"].sum()

        bowling_stats["total_balls"] = (
            bowling_stats["bowlerName"].map(total_balls)
        )

        bowling_stats["total_overs"] = (
            bowling_stats["total_balls"] / 6
        )

    if "runs_conceded" in bowling_stats.columns:

        bowling_stats["economy"] = np.where(
            bowling_stats["total_balls"] > 0,
            bowling_stats["runs_conceded"] /
            bowling_stats["total_balls"] * 6,
            0
        )

    print("\nBowling Statistics")
    display(bowling_stats.head(10))
